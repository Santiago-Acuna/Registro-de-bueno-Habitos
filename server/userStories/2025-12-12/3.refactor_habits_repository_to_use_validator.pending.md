**So that** habit creation/update validates name and icon uniqueness before persisting

**After (Create Method):**
```typescript
async create(data: CreateHabitData): Promise<Habit> {
  // Validate BEFORE any database operation
  const { name, icon } = await this.validator.validateForHabit(
    data.name,
    data.icon
  );

  const createdHabit = await this.prisma.habits.create({
    data: { habitType: this.mapToPrismaHabitType(data.habitType) }
  });

  const globalIdentifier = await this.prisma.globalEntityIdentifiers.create({
    data: {
      name: name.getValue(),  // Validated!
      icon: icon.getValue(),  // Validated!
      entityType: 'habit',
      entityId: createdHabit.id,
    },
  });
  // ... rest of method
}
```

**Update Method Pattern:**
```typescript
async update(id: UUID, habitData: Partial<Habit>): Promise<Habit> {
  // Check if global identifier properties are being updated
  if (habitData.globalIdentifier) {
    const { name, icon } = habitData.globalIdentifier;

    if (name || icon) {
      // Get current values for unchanged properties
      const current = await this.findById(id);
      if (!current) throw new NotFoundError('Habit', id);

      const { name: validatedName, icon: validatedIcon } =
        await this.validator.validateForHabit(
          name?.getValue() || current.globalIdentifier.name.getValue(),
          icon?.getValue() || current.globalIdentifier.icon.getValue(),
          id  // Exclude current habit
        );

      // Update global identifier with validated values
      await this.prisma.globalEntityIdentifiers.update({
        where: { id: current.globalIdentifier.id },
        data: {
          ...(name && { name: validatedName.getValue() }),
          ...(icon && { icon: validatedIcon.getValue() }),
        }
      });
    }
  }

  // Update habit properties as before
  // ...
}
```
