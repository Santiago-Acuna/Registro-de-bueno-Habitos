**After (Create Method):**
```typescript
async create(data: CreateActionTypeData): Promise<ActionType> {
  // Validate habitId first
  if (!data.habitId || data.habitId.trim() === '') {
    throw new Error('habitId is required and must be a valid UUID');
  }

  // Validate BEFORE any database operation
  const { name, icon } = await this.validator.validateForActionType(
    data.name,
    data.icon,
    data.habitId
  );

  const createdActionType = await this.prisma.actionTypes.create({
    data: { habitId: data.habitId }
  });

  const globalIdentifier = await this.prisma.globalEntityIdentifiers.create({
    data: {
      name: name.getValue(),  // Validated!
      icon: icon.getValue(),  // Validated!
      entityType: 'action_type',
      entityId: createdActionType.id,
    },
  });
  // ... rest of method (no catch for P2002 needed)
}
```

**Update Method Pattern:**
```typescript
async update(id: UUID, data: UpdateActionTypeData): Promise<ActionType> {
  // Check if global identifier properties are being updated
  if (data.name || data.icon) {
    // Fetch current action type to get habitId and current values
    const current = await this.findById(id);
    if (!current) throw new NotFoundError(`ActionType with id ${id} not found`);

    const { name: validatedName, icon: validatedIcon } =
      await this.validator.validateForActionType(
        data.name || current.globalIdentifier.name.getValue(),
        data.icon || current.globalIdentifier.icon.getValue(),
        current.habitId,
        id  // Exclude current action type
      );

    // Update global identifier with validated values
    await this.prisma.globalEntityIdentifiers.update({
      where: { entityId: id },
      data: {
        ...(data.name && { name: validatedName.getValue() }),
        ...(data.icon && { icon: validatedIcon.getValue() }),
      }
    });
  }

  // Update action type properties (if any other than name/icon)
  const updateData = { ...data };
  delete updateData.name;
  delete updateData.icon;

  if (Object.keys(updateData).length > 0) {
    try {
      await this.prisma.actionTypes.update({
        where: { id },
        data: updateData as never,
      });
    } catch (error) {
      if (error?.code === 'P2025') {
        throw new NotFoundError(`ActionType with id ${id} not found`);
      }
      throw error;
    }
  }

  return this.findById(id);
}
```

**Key Difference from Habits:**
Action types use `validateForActionType()` which enforces per-habit uniqueness:
- Same name allowed for action types of different habits
- Same name NOT allowed for action types of the same habit
