## Technical Notes

**Integration Test Structure:**
```typescript
describe('Habit Creation with Validation (E2E)', () => {
  it('should create habit with unique name and icon', async () => {
    const response = await request(app.getHttpServer())
      .post('/habits')
      .send({
        name: 'Unique Habit',
        icon: 'https://example.com/unique.png',
        habitType: 'simple'
      })
      .expect(201);

    expect(response.body).toHaveProperty('id');
    expect(response.body.globalIdentifier.name).toBe('Unique Habit');
  });

  it('should reject duplicate habit name', async () => {
    // Create first habit
    await request(app.getHttpServer())
      .post('/habits')
      .send({
        name: 'Duplicate Test',
        icon: 'https://example.com/test1.png',
        habitType: 'simple'
      });

    // Attempt duplicate
    const response = await request(app.getHttpServer())
      .post('/habits')
      .send({
        name: 'Duplicate Test',  // Same name
        icon: 'https://example.com/test2.png',  // Different icon
        habitType: 'simple'
      })
      .expect(409);

    expect(response.body.message).toContain("Habit name 'Duplicate Test' is already in use");
  });
});
```
