**After (Create Method):**
```typescript
async create(data: CreateGlobalEntityIdentifierData): Promise<GlobalEntityIdentifierData> {
  this.logger.log(`Creating global entity identifier: ${data.name} for ${data.entityType}`);

  // Delegate to appropriate validator strategy
  let validated;
  if (data.entityType === 'habit') {
    validated = await this.validator.validateForHabit(data.name, data.icon);
  } else if (data.entityType === 'action_type') {
    // Extract habitId from context (may need to be passed in data)
    validated = await this.validator.validateForActionType(
      data.name,
      data.icon,
      data.habitId  // This field needs to be added to CreateGlobalEntityIdentifierData
    );
  } else {
    throw new ValidationException('Entity type must be one of: habit, action-type');
  }

  const result = await this.repository.create({
    ...data,
    name: validated.name.getValue(),
    icon: validated.icon.getValue(),
  });

  this.logger.log(`Successfully created global entity identifier with id: ${result.id}`);
  return result;
}
```

**Update Method After:**
```typescript
async update(id: UUID, data: UpdateGlobalEntityIdentifierData): Promise<GlobalEntityIdentifierData> {
  const existing = await this.repository.findById(id);
  if (!existing) {
    throw new NotFoundError('Global entity identifier');
  }

  if (data.name || data.icon) {
    const fullName = data.name || existing.name;
    const fullIcon = data.icon || existing.icon;

    let validated;
    if (existing.entityType === 'habit') {
      validated = await this.validator.validateForHabit(
        fullName,
        fullIcon,
        existing.entityId
      );
    } else if (existing.entityType === 'action_type') {
      // Need to get habitId from actionTypes table
      const actionType = await this.getActionTypeById(existing.entityId);
      validated = await this.validator.validateForActionType(
        fullName,
        fullIcon,
        actionType.habitId,
        existing.entityId
      );
    }

    return this.repository.update(id, {
      ...(data.name && { name: validated.name.getValue() }),
      ...(data.icon && { icon: validated.icon.getValue() }),
    });
  }

  return existing;
}
```
